cmake_minimum_required(VERSION 3.16)
project(t150_desk_control LANGUAGES CXX)
include(CTest)
add_subdirectory(lib/desk_protocol)

option(DESK_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)

# One native test executable per source, C++11 with warnings as errors.
function(desk_add_test name source)
  add_executable(${name}_tests ${source})
  target_link_libraries(${name}_tests PRIVATE desk_protocol)
  set_target_properties(${name}_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${name}_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(DESK_SANITIZERS)
      target_compile_options(${name}_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${name}_tests PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME ${name} COMMAND ${name}_tests)
endfunction()

if(BUILD_TESTING)
  desk_add_test(desk_protocol tests/protocol_test.cpp)
  target_compile_definitions(desk_protocol_tests PRIVATE DESK_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  desk_add_test(desk_target tests/target_test.cpp)
  desk_add_test(desk_timer tests/timer_test.cpp)
  desk_add_test(desk_presets tests/presets_test.cpp)
  desk_add_test(desk_identity tests/identity_test.cpp)
  desk_add_test(desk_firmware_export tests/firmware_export_test.cpp)
endif()
