idf_component_register()
# IDF also reads component definitions in script mode to discover dependencies.
if(CMAKE_SCRIPT_MODE_FILE)
  return()
endif()
set(desk_manifest "${PROJECT_DIR}/desk-patches.json")
file(READ "${desk_manifest}" desk_config)
string(JSON desk_count LENGTH "${desk_config}" inputs)
math(EXPR desk_last "${desk_count} - 1")
set(desk_inputs "${desk_manifest}")
foreach(index RANGE 0 ${desk_last})
  string(JSON name GET "${desk_config}" inputs ${index})
  list(APPEND desk_inputs "${PROJECT_DIR}/${name}")
endforeach()
string(JSON desk_controls GET "${desk_config}" controls)
string(JSON desk_maintenance GET "${desk_config}" maintenance)
if(desk_controls OR desk_maintenance)
  foreach(name server_index_v3.h web_server.cpp)
    list(APPEND desk_inputs "${PROJECT_DIR}/src/esphome/components/web_server/${name}")
  endforeach()
endif()
string(JSON desk_ota GET "${desk_config}" ota)
if(desk_ota)
  list(APPEND desk_inputs "${PROJECT_DIR}/src/esphome/components/esphome/ota/ota_esphome.cpp")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${desk_inputs})
execute_process(COMMAND "${PYTHON}" "${PROJECT_DIR}/desk_apply_patches.py" "${PROJECT_DIR}"
                RESULT_VARIABLE desk_result)
if(NOT desk_result EQUAL 0)
  message(FATAL_ERROR "T150 generated-source patch failed")
endif()
